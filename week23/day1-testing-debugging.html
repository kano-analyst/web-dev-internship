🔬 Automated Testing with PHPUnit

Writing Feature Tests

Create Test File

php artisan make:test BookTest

Example Test Cases

// tests/Feature/BookTest.php
namespace Tests\Feature;

use Tests\TestCase;
use App\Models\User;
use App\Models\Book;
use App\Models\Category;
use Illuminate\Foundation\Testing\RefreshDatabase;

class BookTest extends TestCase
{
    use RefreshDatabase;

    /** @test */
    public function authenticated_users_can_view_books()
    {
        $user = User::factory()->create();
        
        $response = $this->actingAs($user)->get('/books');
        
        $response->assertStatus(200);
        $response->assertViewIs('books.index');
    }

    /** @test */
    public function users_can_create_a_book()
    {
        $user = User::factory()->create(['is_admin' => true]);
        $category = Category::factory()->create();
        
        $bookData = [
            'title' => 'Test Book',
            'author' => 'Test Author',
            'isbn' => '1234567890',
            'category_id' => $category->id,
            'quantity' => 5,
            'description' => 'Test description'
        ];
        
        $response = $this->actingAs($user)
                         ->post('/books', $bookData);
        
        $this->assertDatabaseHas('books', [
            'title' => 'Test Book',
            'author' => 'Test Author'
        ]);
        
        $response->assertRedirect('/books');
    }

    /** @test */
    public function book_title_is_required()
    {
        $user = User::factory()->create(['is_admin' => true]);
        $category = Category::factory()->create();
        
        $response = $this->actingAs($user)
                         ->post('/books', [
                             'title' => '',
                             'author' => 'Test Author',
                             'isbn' => '1234567890',
                             'category_id' => $category->id,
                             'quantity' => 5
                         ]);
        
        $response->assertSessionHasErrors('title');
    }

    /** @test */
    public function users_can_search_books()
    {
        $user = User::factory()->create();
        $category = Category::factory()->create();
        
        Book::factory()->create([
            'title' => 'Laravel Tutorial',
            'category_id' => $category->id
        ]);
        
        Book::factory()->create([
            'title' => 'PHP Basics',
            'category_id' => $category->id
        ]);
        
        $response = $this->actingAs($user)
                         ->get('/books?search=Laravel');
        
        $response->assertSee('Laravel Tutorial');
        $response->assertDontSee('PHP Basics');
    }

    /** @test */
    public function book_availability_decreases_when_borrowed()
    {
        $user = User::factory()->create();
        $category = Category::factory()->create();
        $book = Book::factory()->create([
            'category_id' => $category->id,
            'quantity' => 5,
            'available_quantity' => 5
        ]);
        
        $this->actingAs($user)
             ->post('/borrowings', [
                 'book_id' => $book->id,
                 'user_id' => $user->id
             ]);
        
        $book->refresh();
        $this->assertEquals(4, $book->available_quantity);
    }
}

php artisan test

🐛 Debugging Techniques

Laravel Debugging Tools

1. dd() - Dump and Die

// Dump variable and stop execution
dd($books);

// Dump multiple variables
dd($books, $categories, $user);

2. dump() - Dump Without Stopping

🔧 Common Bugs and Solutions

Bug Fix Examples

Bug 1: N+1 Query Problem

// Problem: This generates 101 queries (1 + 100)
$borrowings = Borrowing::all(); // 1 query
foreach ($borrowings as $borrowing) {
    echo $borrowing->book->title; // 100 queries
}

// Solution: Use eager loading
$borrowings = Borrowing::with(['book', 'user'])->get(); // 3 queries total

Bug 2: Mass Assignment Error

// Problem: "Add [field] to fillable property"
Book::create($request->all()); // Error!

// Solution: Add to $fillable array in model
protected $fillable = [
    'title', 'author', 'isbn', 'category_id', 
    'description', 'quantity', 'available_quantity'
];

Bug 3: File Upload Not Working

// Problem: Images not displaying

// Solution 1: Create symbolic link
php artisan storage:link

// Solution 2: Update form to include enctype
<form method="POST" enctype="multipart/form-data">

// Solution 3: Check file validation
$request->validate([
    'cover_image' => 'nullable|image|mimes:jpeg,png,jpg|max:2048'
]);

💡 Testing Best Practices:
  • Test like a real user would use the application
  • Try to break the application with unusual inputs
  • Test edge cases (empty data, very long text, special characters)
  • Test both success and failure scenarios
  • Document steps to reproduce bugs
  • Retest after fixing bugs to ensure fix works
