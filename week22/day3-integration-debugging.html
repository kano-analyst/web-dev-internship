🔗 Step 1: Complete Borrowing System

BorrowingController Implementation

// app/Http/Controllers/BorrowingController.php
namespace App\Http\Controllers;

use App\Models\Borrowing;
use App\Models\Book;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use App\Notifications\BookBorrowedNotification;
use App\Notifications\BookReturnReminderNotification;

class BorrowingController extends Controller
{
    public function index()
    {
        $borrowings = Borrowing::with(['book', 'user'])
                               ->when(!Auth::user()->is_admin, function($query) {
                                   $query->where('user_id', Auth::id());
                               })
                               ->latest()
                               ->paginate(15);
        
        return view('borrowings.index', compact('borrowings'));
    }

    public function create()
    {
        $availableBooks = Book::where('available_quantity', '>', 0)->get();
        return view('borrowings.create', compact('availableBooks'));
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'book_id' => 'required|exists:books,id',
            'user_id' => 'required|exists:users,id',
            'notes' => 'nullable|string|max:500'
        ]);

        $book = Book::findOrFail($validated['book_id']);

        // Check availability
        if ($book->available_quantity < 1) {
            return back()->with('error', 'This book is currently unavailable.');
        }

        // Create borrowing record
        $borrowing = Borrowing::create([
            'user_id' => $validated['user_id'],
            'book_id' => $validated['book_id'],
            'borrowed_date' => now(),
            'due_date' => now()->addDays(14), // 14 days loan period
            'status' => 'borrowed',
            'notes' => $validated['notes']
        ]);

        // Update book availability
        $book->decrement('available_quantity');

        // Send notification
        $borrowing->user->notify(new BookBorrowedNotification($borrowing));

        return redirect()->route('borrowings.index')
                        ->with('success', 'Book borrowed successfully!');
    }

    public function return(Borrowing $borrowing)
    {
        // Check authorization
        if (!Auth::user()->is_admin && $borrowing->user_id !== Auth::id()) {
            abort(403, 'Unauthorized action.');
        }

        // Update borrowing
        $borrowing->update([
            'return_date' => now(),
            'status' => 'returned'
        ]);

        // Update book availability
        $borrowing->book->increment('available_quantity');

        // Check for overdue and create fine if necessary
        if ($borrowing->return_date->isAfter($borrowing->due_date)) {
            $daysOverdue = $borrowing->return_date->diffInDays($borrowing->due_date);
            $fineAmount = $daysOverdue * 10; // 10 naira per day

            $borrowing->fine()->create([
                'amount' => $fineAmount,
                'status' => 'pending'
            ]);
        }

        return redirect()->route('borrowings.index')
                        ->with('success', 'Book returned successfully!');
    }

    public function destroy(Borrowing $borrowing)
    {
        // Only allow deletion if not returned
        if ($borrowing->status === 'borrowed') {
            $borrowing->book->increment('available_quantity');
        }

        $borrowing->delete();

        return redirect()->route('borrowings.index')
                        ->with('success', 'Borrowing record deleted.');
    }
}

Book Borrowed Notification

// app/Notifications/BookBorrowedNotification.php
namespace App\Notifications;

use Illuminate\Notifications\Notification;
use Illuminate\Notifications\Messages\MailMessage;
use App\Models\Borrowing;

class BookBorrowedNotification extends Notification
{
    protected $borrowing;

    public function __construct(Borrowing $borrowing)
    {
        $this->borrowing = $borrowing;
    }

    public function via($notifiable)
    {
        return ['mail'];
    }

    public function toMail($notifiable)
    {
        return (new MailMessage)
            ->subject('Book Borrowed Successfully')
            ->greeting('Hello ' . $notifiable->name . '!')
            ->line('You have successfully borrowed the following book:')
            ->line('**Book:** ' . $this->borrowing->book->title)
            ->line('**Author:** ' . $this->borrowing->book->author)
            ->line('**Borrowed Date:** ' . $this->borrowing->borrowed_date->format('M d, Y'))
            ->line('**Due Date:** ' . $this->borrowing->due_date->format('M d, Y'))
            ->line('Please return the book on or before the due date to avoid fines.')
            ->action('View My Borrowings', route('borrowings.index'))
            ->line('Thank you for using our library!');
    }
}

Schedule Reminder Emails

// app/Console/Kernel.php
protected function schedule(Schedule $schedule)
{
    // Send reminder emails 2 days before due date
    $schedule->call(function () {
        $upcomingDue = Borrowing::where('status', 'borrowed')
                                ->whereDate('due_date', now()->addDays(2))
                                ->get();

        foreach ($upcomingDue as $borrowing) {
            $borrowing->user->notify(new BookReturnReminderNotification($borrowing));
        }
    })->daily();

    // Mark overdue borrowings
    $schedule->call(function () {
        Borrowing::where('status', 'borrowed')
                ->whereDate('due_date', '<', now())
                ->update(['status' => 'overdue']);
    })->daily();
}

🐛 Step 3: Common Bugs & Fixes

Bug Tracking Checklist

Issue Symptom Solution
Image not displaying Broken image icons Run php artisan storage:link
Validation errors not showing Form submits without feedback Add @error directives in Blade
Relationship not loading N+1 query problem Use ->with(['relation']) eager loading
Pagination links broken Page numbers don't preserve filters Use ->appends(request()->query())
Old input not retained Form clears on validation error Use old('field_name') in inputs
Unauthorized access Users can access restricted pages Add middleware or authorization checks

Fixing Search Functionality

// Update BookController@index with search
public function index(Request $request)
{
    $query = Book::with('category');

    // Search filter
    if ($request->filled('search')) {
        $search = $request->search;
        $query->where(function($q) use ($search) {
            $q->where('title', 'like', "%{$search}%")
              ->orWhere('author', 'like', "%{$search}%")
              ->orWhere('isbn', 'like', "%{$search}%");
        });
    }

    // Category filter
    if ($request->filled('category')) {
        $query->where('category_id', $request->category);
    }

    // Availability filter
    if ($request->filled('available')) {
        if ($request->available === 'yes') {
            $query->where('available_quantity', '>', 0);
        } else {
            $query->where('available_quantity', '=', 0);
        }
    }

    $books = $query->latest()->paginate(12)->appends($request->query());
    $categories = Category::all();

    return view('books.index', compact('books', 'categories'));
}

🚀 Step 4: Performance Optimization

Optimization Techniques

1. Database Query Optimization

// Bad - N+1 query problem
$borrowings = Borrowing::all();
foreach ($borrowings as $borrowing) {
    echo $borrowing->book->title; // Separate query each time
}

// Good - Eager loading
$borrowings = Borrowing::with(['book', 'user'])->get();
foreach ($borrowings as $borrowing) {
    echo $borrowing->book->title; // No extra queries
}

2. Caching Common Data

// Cache categories for 1 hour

    return Category::all();
